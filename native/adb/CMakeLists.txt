cmake_minimum_required(VERSION 3.22)
project(vegaadb LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_NO_ASM ON CACHE BOOL "" FORCE)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_subdirectory(../../third_party/adb/vendor/boringssl boringssl EXCLUDE_FROM_ALL)
add_library(vegaadb_core STATIC crypto.cpp transport.cpp)
target_include_directories(vegaadb_core PUBLIC . ../../third_party/adb/vendor/boringssl/include)
target_link_libraries(vegaadb_core PUBLIC ssl crypto)
if(ANDROID)
  target_link_libraries(vegaadb_core PUBLIC android)
elseif(WIN32)
  target_link_libraries(vegaadb_core PUBLIC ws2_32)
elseif(NOT ANDROID)
  target_link_libraries(vegaadb_core PUBLIC pthread)
endif()
if(ANDROID)
  add_library(vegaadb SHARED jni.cpp)
  target_link_libraries(vegaadb PRIVATE vegaadb_core)
  target_link_options(vegaadb PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,-z,noexecstack -Wl,--exclude-libs,ALL -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map)
else()
  enable_testing()
  add_library(aosp_pairing_reference STATIC
    ../../third_party/adb/reference/pairing_auth/pairing_auth.cpp
    ../../third_party/adb/reference/pairing_auth/aes_128_gcm.cpp)
  target_include_directories(aosp_pairing_reference PRIVATE test_shims
    ../../third_party/adb/reference/pairing_auth/include)
  target_link_libraries(aosp_pairing_reference PUBLIC crypto)
  add_executable(vegaadb_tests tests.cpp)
  target_include_directories(vegaadb_tests PRIVATE ../../third_party/adb/reference/pairing_auth/include)
  target_link_libraries(vegaadb_tests PRIVATE vegaadb_core aosp_pairing_reference)
  add_test(NAME adb_crypto_transport COMMAND vegaadb_tests)
endif()
